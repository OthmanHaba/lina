# Alerts & Notifications Implementation Plan (M2 Slice 1)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Threshold and missing-logs alerts evaluated every minute against Loki, with state-transition notifications over email, webhook, and Telegram, managed from a team-scoped Alerts page.

**Architecture:** A scheduler-driven `alerts:evaluate` command runs each enabled alert as one Loki instant query (`sum(count_over_time(...))`) with the team's OrgID, flips `ok`/`firing` state on breach/recovery, appends an `alert_events` row, and hands transitions to an `AlertNotifier` that delivers to each attached channel synchronously with per-channel failure isolation and auto-disable after 10 consecutive failures.

**Tech Stack:** Laravel 13 (scheduler, Mailable, Http client), existing `LogQlBuilder`, Inertia v3 + React 19 + Wayfinder (`--with-form`), Pest 4.

**Spec:** `docs/superpowers/specs/2026-07-26-alerts-design.md`

## Global Constraints

- Tenancy: evaluation queries derive `X-Scope-OrgID` exclusively from the alert's `team_id` (`team-{id}`); alert/channel routes live under `{current_team}` with the guard `abort_unless($model->team_id === $request->user()->current_team_id, 404)`.
- Repo quirk: controller methods for routes under `{current_team}` MUST accept `string $current_team` positionally before route-bound models (see `ServerController::show`).
- Notification semantics: send ONLY on state transitions (`ok→firing`, `firing→ok`); never while state is unchanged.
- Channel auto-disable: 10 consecutive failures → `enabled=false`; any success resets `failure_count` to 0.
- Delivery is synchronous with per-send try/catch and short HTTP timeouts (5s); a channel failure never affects other channels or evaluation.
- Loki unreachable/erroring during evaluation → skip that alert this round: no state change, no notification, log a warning.
- Alert config shape (canonical): `{servers: int[], sources: string[], search: string|null, threshold: int|null, window_minutes: int}`; empty `servers`/`sources` = all; `search`/`threshold` only for `threshold` type.
- Channel config shapes: email `{address}`, webhook `{url}`, telegram `{chat_id}`.
- Telegram uses one platform bot: `config('services.telegram.bot_token')` from env `TELEGRAM_BOT_TOKEN`.
- Wayfinder regeneration in this repo MUST use `php artisan wayfinder:generate --no-interaction --with-form`.
- Run `vendor/bin/pint --dirty --format agent` before every commit touching PHP; `php artisan test --compact --filter=<name>` for minimal test runs.

```php
<?php

namespace App\Models;

use App\Enums\ChannelType;
use Database\Factories\NotificationChannelFactory;
use Illuminate\Database\Eloquent\Attributes\Fillable;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

/**
 * @property int $id
 * @property int $team_id
 * @property ChannelType $type
 * @property string $name
 * @property array<string, mixed> $config
 * @property bool $enabled
 * @property int $failure_count
 * @property-read Team $team
 */
#[Fillable(['team_id', 'type', 'name', 'config', 'enabled', 'failure_count'])]
class NotificationChannel extends Model
{
    /** @use HasFactory<NotificationChannelFactory> */
    use HasFactory;

    /**
     * Consecutive delivery failures before a channel is auto-disabled.
     */
    public const int MAX_FAILURES = 10;

    /**
     * Record a delivery failure; disable the channel after MAX_FAILURES in a row.
     */
    public function recordFailure(): void
    {
        $this->increment('failure_count');

        if ($this->failure_count >= self::MAX_FAILURES) {
            $this->forceFill(['enabled' => false])->save();
        }
    }

    /**
     * Record a delivery success, resetting the failure streak.
     */
    public function recordSuccess(): void
    {
        if ($this->failure_count > 0) {
            $this->forceFill(['failure_count' => 0])->save();
        }
    }

    /** @return BelongsTo<Team, $this> */
    public function team(): BelongsTo
    {
        return $this->belongsTo(Team::class);
    }

    /** @return array<string, string> */
    protected function casts(): array
    {
        return [
            'type' => ChannelType::class,
            'config' => 'array',
            'enabled' => 'boolean',
            'failure_count' => 'integer',
        ];
    }
}
```

```php
namespace App\Http\Requests\Alerts;

use App\Enums\ChannelType;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Validation\Rule;

class SaveChannelRequest extends FormRequest
{
    /** @return array<string, mixed> */
    public function rules(): array
    {
        $type = ChannelType::tryFrom((string) $this->input('type'));

        return [
            'type' => ['required', Rule::enum(ChannelType::class)],
            'name' => ['required', 'string', 'max:100'],
            'enabled' => ['sometimes', 'boolean'],
            'config' => ['required', 'array'],
            ...match ($type) {
                ChannelType::Email => ['config.address' => ['required', 'email', 'max:255']],
                ChannelType::Webhook => ['config.url' => ['required', 'url:http,https', 'max:1024']],
                ChannelType::Telegram => ['config.chat_id' => ['required', 'string', 'max:64']],
                default => [],
            },
        ];
    }
}
```

```php
<?php

namespace App\Http\Controllers\Alerts;

use App\Http\Controllers\Controller;
use App\Http\Requests\Alerts\SaveChannelRequest;
use App\Models\NotificationChannel;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Inertia\Inertia;

class NotificationChannelController extends Controller
{
    /**
     * Add a notification channel to the current team.
     */
    public function store(SaveChannelRequest $request, string $current_team): RedirectResponse
    {
        NotificationChannel::create([
            'team_id' => $request->user()->current_team_id,
            'type' => $request->validated('type'),
            'name' => $request->validated('name'),
            'config' => $request->validated('config'),
            'enabled' => $request->validated('enabled', true),
        ]);

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Channel added.')]);

        return back();
    }

    /**
     * Update a channel. Re-enabling (or any edit) clears the failure streak so a
     * fixed endpoint gets a fresh start.
     */
    public function update(SaveChannelRequest $request, string $current_team, NotificationChannel $channel): RedirectResponse
    {
        $this->authorizeChannel($request, $channel);

        $channel->update([
            'type' => $request->validated('type'),
            'name' => $request->validated('name'),
            'config' => $request->validated('config'),
            'enabled' => $request->validated('enabled', true),
            'failure_count' => 0,
        ]);

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Channel updated.')]);

        return back();
    }

    /**
     * Delete a channel; pivot rows cascade, alerts remain.
     */
    public function destroy(Request $request, string $current_team, NotificationChannel $channel): RedirectResponse
    {
        $this->authorizeChannel($request, $channel);

        $channel->delete();

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Channel removed.')]);

        return back();
    }

    /**
     * Channels are only visible within their own team.
     */
    protected function authorizeChannel(Request $request, NotificationChannel $channel): void
    {
        abort_unless($channel->team_id === $request->user()->current_team_id, 404);
    }
}
```

```php
<?php

namespace App\Http\Controllers\Alerts;

use App\Http\Controllers\Controller;
use App\Http\Requests\Alerts\SaveAlertRequest;
use App\Models\Alert;
use App\Models\AlertEvent;
use App\Models\LogSource;
use App\Models\NotificationChannel;
use App\Models\Server;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Inertia\Inertia;
use Inertia\Response;

class AlertController extends Controller
{
    /**
     * The alerts page: alerts, channels, and the filter options they share with logs.
     */
    public function index(Request $request): Response
    {
        $teamId = $request->user()->current_team_id;

        return Inertia::render('alerts/index', [
            'alerts' => Alert::where('team_id', $teamId)
                ->with(['channels:id', 'events' => fn ($query) => $query->latest()->limit(10)])
                ->latest()
                ->get()
                ->map(fn (Alert $alert) => [
                    'id' => $alert->id,
                    'name' => $alert->name,
                    'type' => $alert->type->value,
                    'state' => $alert->state->value,
                    'enabled' => $alert->enabled,
                    'config' => $alert->config,
                    'channelIds' => $alert->channels->pluck('id'),
                    'lastStateChangeAt' => $alert->last_state_change_at?->toISOString(),
                    'events' => $alert->events->map(fn (AlertEvent $event) => [
                        'state' => $event->state->value,
                        'measuredValue' => $event->measured_value,
                        'createdAt' => $event->created_at->toISOString(),
                    ]),
                ]),
            'channels' => NotificationChannel::where('team_id', $teamId)
                ->orderBy('name')
                ->get()
                ->map(fn (NotificationChannel $channel) => [
                    'id' => $channel->id,
                    'type' => $channel->type->value,
                    'name' => $channel->name,
                    'enabled' => $channel->enabled,
                    'failureCount' => $channel->failure_count,
                    'config' => $channel->config,
                ]),
            'servers' => Server::where('team_id', $teamId)
                ->get(['id', 'name'])
                ->map(fn (Server $server) => ['id' => $server->id, 'name' => $server->name]),
            'sources' => LogSource::whereIn('server_id', Server::where('team_id', $teamId)->select('id'))
                ->where('enabled', true)
                ->distinct()
                ->pluck('name'),
        ]);
    }

    /**
     * Create an alert with attached channels.
     */
    public function store(SaveAlertRequest $request, string $current_team): RedirectResponse
    {
        $alert = Alert::create([
            'team_id' => $request->user()->current_team_id,
            'name' => $request->validated('name'),
            'type' => $request->validated('type'),
            'config' => $request->alertConfig(),
            'enabled' => $request->validated('enabled', true),
        ]);

        $alert->channels()->sync($request->validated('channels', []));

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Alert created.')]);

        return back();
    }

    /**
     * Update an alert and its channels.
     */
    public function update(SaveAlertRequest $request, string $current_team, Alert $alert): RedirectResponse
    {
        $this->authorizeAlert($request, $alert);

        $alert->update([
            'name' => $request->validated('name'),
            'type' => $request->validated('type'),
            'config' => $request->alertConfig(),
            'enabled' => $request->validated('enabled', true),
        ]);

        $alert->channels()->sync($request->validated('channels', []));

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Alert updated.')]);

        return back();
    }

    /**
     * Delete an alert; events and pivot rows cascade.
     */
    public function destroy(Request $request, string $current_team, Alert $alert): RedirectResponse
    {
        $this->authorizeAlert($request, $alert);

        $alert->delete();

        Inertia::flash('toast', ['type' => 'success', 'message' => __('Alert deleted.')]);

        return back();
    }

    /**
     * Alerts are only visible within their own team.
     */
    protected function authorizeAlert(Request $request, Alert $alert): void
    {
        abort_unless($alert->team_id === $request->user()->current_team_id, 404);
    }
}
```

`app/Services/AlertNotifier.php`:

```php
<?php

namespace App\Services;

use App\Enums\ChannelType;
use App\Mail\AlertStateChangedMail;
use App\Models\Alert;
use App\Models\AlertEvent;
use App\Models\NotificationChannel;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Mail;
use Throwable;

class AlertNotifier
{
    /**
     * Deliver a state-change to every enabled channel on the alert. Each channel
     * is isolated: one failing endpoint never blocks the others or the evaluator.
     */
    public function notify(Alert $alert, AlertEvent $event): void
    {
        foreach ($alert->channels()->where('enabled', true)->get() as $channel) {
            try {
                match ($channel->type) {
                    ChannelType::Email => $this->sendEmail($channel, $alert, $event),
                    ChannelType::Webhook => $this->sendWebhook($channel, $alert, $event),
                    ChannelType::Telegram => $this->sendTelegram($channel, $alert, $event),
                };

                $channel->recordSuccess();
            } catch (Throwable $exception) {
                $channel->recordFailure();
                Log::warning('Alert notification failed', [
                    'alert_id' => $alert->id,
                    'channel_id' => $channel->id,
                    'error' => $exception->getMessage(),
                ]);
            }
        }
    }

    /**
     * Deep link to the logs page pre-filtered to the alert's scope.
     */
    public function deepLink(Alert $alert): string
    {
        $query = array_filter([
            'servers' => implode(',', $alert->config['servers'] ?? []),
            'sources' => implode(',', $alert->config['sources'] ?? []),
            'search' => $alert->config['search'] ?? '',
        ]);

        $url = route('logs.index', ['current_team' => $alert->team->slug]);

        return $query === [] ? $url : $url.'?'.http_build_query($query);
    }

    protected function sendEmail(NotificationChannel $channel, Alert $alert, AlertEvent $event): void
    {
        Mail::to($channel->config['address'])
            ->send(new AlertStateChangedMail($alert, $event, $this->deepLink($alert)));
    }

    protected function sendWebhook(NotificationChannel $channel, Alert $alert, AlertEvent $event): void
    {
        Http::timeout(5)->post($channel->config['url'], [
            'alert' => ['id' => $alert->id, 'name' => $alert->name, 'type' => $alert->type->value],
            'state' => $event->state->value,
            'measured_value' => $event->measured_value,
            'threshold' => $alert->config['threshold'],
            'window_minutes' => $alert->config['window_minutes'],
            'servers' => $alert->config['servers'],
            'sources' => $alert->config['sources'],
            'search' => $alert->config['search'],
            'fired_at' => $event->created_at->toISOString(),
            'url' => $this->deepLink($alert),
        ])->throw();
    }

    protected function sendTelegram(NotificationChannel $channel, Alert $alert, AlertEvent $event): void
    {
        $token = config('services.telegram.bot_token');

        throw_if(blank($token), new \RuntimeException('TELEGRAM_BOT_TOKEN is not configured.'));

        $text = sprintf(
            "<b>%s</b> is now <b>%s</b>\n%d line(s) in the last %d minute(s)\n%s",
            e($alert->name),
            $event->state->value,
            $event->measured_value,
            $alert->config['window_minutes'],
            $this->deepLink($alert),
        );

        Http::timeout(5)->post("https://api.telegram.org/bot{$token}/sendMessage", [
            'chat_id' => $channel->config['chat_id'],
            'text' => $text,
            'parse_mode' => 'HTML',
        ])->throw();
    }
}
```

**Files:**

**Interfaces:**

- Produces: `AlertEvaluator::evaluate(Alert $alert): void` — runs one instant query, applies transition rules, appends events, calls the notifier. Task 6's command calls exactly this. Behavior contract:

  - Query: Loki `GET /loki/api/v1/query` with `query = sum(count_over_time(<selector> [<W>m]))` and header `X-Scope-OrgID: team-{team_id}`; count = `data.result[0].value[1]` cast to int, `0` when the vector is empty.
  - Breach: threshold → `count > threshold`; missing_logs → `count === 0`.
  - Transitions only on change; each transition sets `state`, `last_state_change_at`, creates an `AlertEvent` with the entered state + measured count, and notifies.
  - Loki connection/HTTP failure → log warning, bump nothing, return (no state change, no notification).
  - Always bumps `last_evaluated_at` on completed (non-errored) evaluations.

`app/Services/AlertEvaluator.php`:

```php
<?php

namespace App\Services;

use App\Enums\AlertState;
use App\Enums\AlertType;
use App\Models\Alert;
use App\Models\Server;
use Illuminate\Http\Client\ConnectionException;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class AlertEvaluator
{
    public function __construct(
        private LogQlBuilder $builder,
        private AlertNotifier $notifier,
    ) {}

    /**
     * Evaluate one alert: count matching lines over the window and apply
     * state-transition rules. Errors skip the round; they never throw.
     */
    public function evaluate(Alert $alert): void
    {
        $teamServerIds = Server::where('team_id', $alert->team_id)->pluck('id')->all();
        $configured = $alert->config['servers'] ?? [];
        $serverIds = $configured === []
            ? $teamServerIds
            : array_values(array_intersect(array_map(intval(...), $configured), $teamServerIds));

        if ($serverIds === []) {
            $alert->forceFill(['last_evaluated_at' => now()])->save();

            return;
        }

        $selector = $this->builder->build(
            $serverIds,
            $alert->config['sources'] ?? [],
            $alert->type === AlertType::Threshold ? ($alert->config['search'] ?? null) : null,
        );

        $window = (int) $alert->config['window_minutes'];
        $query = 'sum(count_over_time('.$selector.' ['.$window.'m]))';

        try {
            $response = Http::withHeaders(['X-Scope-OrgID' => 'team-'.$alert->team_id])
                ->timeout(10)
                ->get(rtrim(config('services.loki.url'), '/').'/loki/api/v1/query', ['query' => $query]);
        } catch (ConnectionException $exception) {
            Log::warning('Alert evaluation skipped: Loki unreachable', ['alert_id' => $alert->id, 'error' => $exception->getMessage()]);

            return;
        }

        if ($response->failed()) {
            Log::warning('Alert evaluation skipped: Loki error', ['alert_id' => $alert->id, 'status' => $response->status()]);

            return;
        }

        $count = (int) ($response->json('data.result.0.value.1') ?? 0);

        $breached = $alert->type === AlertType::Threshold
            ? $count > (int) $alert->config['threshold']
            : $count === 0;

        $newState = match (true) {
            $breached && $alert->state === AlertState::Ok => AlertState::Firing,
            ! $breached && $alert->state === AlertState::Firing => AlertState::Ok,
            default => null,
        };

        $alert->forceFill(['last_evaluated_at' => now()]);

        if ($newState === null) {
            $alert->save();

            return;
        }

        $alert->forceFill(['state' => $newState, 'last_state_change_at' => now()])->save();

        $event = $alert->events()->create(['state' => $newState, 'measured_value' => $count]);

        $this->notifier->notify($alert, $event);
    }
}
```

```php
<?php

namespace App\Console\Commands;

use App\Models\Alert;
use App\Services\AlertEvaluator;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Log;
use Throwable;

class EvaluateAlerts extends Command
{
    protected $signature = 'alerts:evaluate';

    protected $description = 'Evaluate all enabled alerts against Loki and send state-change notifications';

    public function handle(AlertEvaluator $evaluator): int
    {
        Alert::where('enabled', true)
            ->with('team')
            ->get()
            ->each(function (Alert $alert) use ($evaluator) {
                try {
                    $evaluator->evaluate($alert);
                } catch (Throwable $exception) {
                    Log::error('Alert evaluation crashed', ['alert_id' => $alert->id, 'error' => $exception->getMessage()]);
                }
            });

        return self::SUCCESS;
    }
}
```
